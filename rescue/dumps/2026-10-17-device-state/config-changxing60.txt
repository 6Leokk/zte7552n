### /etc/changxing60/voice-audio.conf
# Calibration for changxing60-voice-audio. Every key here is optional; with this file absent or
# fully commented out the built-in defaults are used. Copy a line, uncomment it, edit the number.
#
# This file exists because the one thing in the voice path that is NOT the driver's is the table of
# DSP mode numbers. A profile Select word is (mode << 24) | (mode << 16) | 5, where the mode indexes
# a parameter set inside /lib/firmware/audio_structure and 5 is VBC_DAI_ID_VOICE. The numbering
# belongs to this machine's parameter table, so it cannot be read out of the driver and it cannot be
# proved from the blob alone -- it has to be confirmed by listening to a real call.
#
# The shipped defaults are the VIPME M8 table (same UMS9620/SC2730 family) at the indices where our
# own blob's gain curves look like a per-band table should. selftest prints those curves. To find
# the right number for a band: place a call, listen, change the key, restart the call, listen again.
# Band 0/1/2/3 are narrowband/wideband/super-wideband/fullband.
#
# mode-handset-nb   = 0
# mode-handset-wb   = 2
# mode-handset-swb  = 4
# mode-handset-fb   = 5
# mode-handsfree-nb = 7
# mode-handsfree-wb = 9
# mode-handsfree-swb = 11
# mode-handsfree-fb = 12
# mode-bluetooth-nb = 28
# mode-bluetooth-wb = 30
# mode-bluetooth-swb = 32
# mode-bluetooth-fb = 33

# Analog codec gains. These are the three knobs that decide how loud the call is before the DSP gets
# a say, and nothing has ever set them on this board: the card comes up with DAC 0, AO 0 and EAR 15.
# An analog stage that is too loud clips before any DSP attenuation can help, and one that is too
# quiet cannot be recovered downstream. The values below are e5's fallbacks, and the card's own
# limits are enforced on top of them (DAC Gain DAC Playback Volume is 0..2 here, not 0..3, so the
# default of 3 is written as 2 and the clamp is logged).
#
# gain-dac-receiver = 3
# gain-dac-speaker  = 3
# gain-ao-receiver  = 6
# gain-ao-speaker   = 4
# gain-ear-nb       = 2
# gain-ear-wb       = 2
